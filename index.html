Backend Google Apps Script (GAS Engine)

Kode ini menangani pembuatan database di Google Sheets, pengiriman WA, dan penampung data RSVP otomatis (DoPost Webhook).

1

Buat Google Sheets

Buka Google Sheets baru di Google Drive Anda.

2

Tempel Kode Ke Apps Script

Klik Ekstensi > Apps Script. Paste seluruh kode di bawah ini.

3

Jalankan Setup Database

Pilih fungsi setupDatabaseSheet lalu klik Run untuk membuat struktur kolom & menu otomatis.

/**
 * ==============================================================================
 * INVITASYSTEM PRO - BACKEND ENGINE (GOOGLE APPS SCRIPT)
 * ==============================================================================
 * Fitur: 
 * 1. Auto Database Setup & Menu Custom di Google Sheets
 * 2. Auto Dynamic URL & WhatsApp Message Formula
 * 3. Kirim WA Langsung via Cell Selection
 * 4. Webhook Web App Endpoint (doPost) untuk Penampungan Form RSVP
 */

// URL UTAMA DOMAIN WEBSITE UNDANGAN DIGITAL ANDA
var BASE_URL_UNDANGAN = "https://undangan-kita.com/manten-bagus-rara";

/**
 * 1. MEMBUAT MENU KUSTOM DI GOOGLE SHEETS
 */
function onOpen() {
  var ui = SpreadsheetApp.getUi();
  ui.createMenu('💌 Kirim Undangan Pro')
    .addItem('🚀 Kirim WA Tamu (Baris Dipilih)', 'kirimWAPilihan')
    .addSeparator()
    .addItem('⚙️ Initial Setup Database Sheet', 'setupDatabaseSheet')
    .addToUi();
}

/**
 * 2. FUNSI SETUP OTOMATIS STRUKTUR DATABASE
 */
function setupDatabaseSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getActiveSheet();
  sheet.setName("Daftar Tamu");

  // Header Utama
  var headers = [["Nama Tamu", "No WhatsApp", "Kategori", "Link Undangan Dinamis", "Draf Pesan WA", "Status Kirim", "RSVP Konfirmasi", "Ucapan & Doa"]];
  sheet.getRange(1, 1, 1, 8).setValues(headers);
  
  // Style Header
  var headerRange = sheet.getRange(1, 1, 1, 8);
  headerRange.setFontWeight("bold")
             .setBackground("#312e81")
             .setFontColor("#ffffff")
             .setVerticalAlignment("middle")
             .setHorizontalAlignment("center");
  sheet.setRowHeight(1, 35);

  // Set Formula Otomatis Baris 2 sampai 100
  for (var i = 2; i <= 100; i++) {
    sheet.getRange(i, 4).setFormula('=IF(A' + i + '<>""; "' + BASE_URL_UNDANGAN + '?to=" & ENCODEURL(A' + i + '); "")');
    sheet.getRange(i, 5).setFormula('=IF(A' + i + '<>""; "Kepada Yth. " & A' + i + ' & "%0A%0ATanpa mengurangi rasa hormat, kami mengundang Bapak/Ibu/Saudara/i untuk hadir di acara pernikahan kami.%0A%0ADetail Undangan:%0A" & D' + i + ' & "%0A%0ATerima Kasih."; "")');
  }

  // Set Auto Column Widths
  sheet.setColumnWidth(1, 200); // Nama
  sheet.setColumnWidth(2, 140); // WA
  sheet.setColumnWidth(3, 120); // Kategori
  sheet.setColumnWidth(4, 300); // Link
  sheet.setColumnWidth(5, 300); // Pesan
  sheet.setColumnWidth(6, 130); // Status
  sheet.setColumnWidth(7, 140); // RSVP
  sheet.setColumnWidth(8, 250); // Ucapan

  SpreadsheetApp.getUi().alert("✅ Database Spreadsheet Berhasil Disiapkan!");
}

/**
 * 3. FUNGSI KLIK KIRIM WA DARI GOOGLE SHEETS
 */
function kirimWAPilihan() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var cell = sheet.getActiveCell();
  var row = cell.getRow();
  
  if (row < 2) {
    SpreadsheetApp.getUi().alert("⚠️ Silakan pilih baris data tamu (bukan baris header)!");
    return;
  }
  
  var nama = sheet.getRange(row, 1).getValue();
  var noHp = sheet.getRange(row, 2).getValue();
  var pesan = sheet.getRange(row, 5).getValue();
  
  if (!nama || !noHp) {
    SpreadsheetApp.getUi().alert("⚠️ Nama Tamu dan Nomor WhatsApp harus diisi!");
    return;
  }
  
  // Clean & Format Nomor HP ke Kode Negara (62)
  noHp = String(noHp).replace(/[^0-9]/g, '');
  if (noHp.startsWith("0")) {
    noHp = "62" + noHp.substring(1);
  }
  
  var waUrl = "https://api.whatsapp.com/send?phone=" + noHp + "&text=" + pesan;
  
  // Update Status Kirim di Spreadsheet
  sheet.getRange(row, 6).setValue("SUDAH DIKIRIM")
                        .setBackground("#d1fae5")
                        .setFontColor("#065f46")
                        .setFontWeight("bold");
  
  // Buka Tab WhatsApp Web
  var html = '<script>window.open("' + waUrl + '", "_blank");google.script.host.close();</script>';
  var userInterface = HtmlService.createHtmlOutput(html).setHeight(50).setWidth(200);
  SpreadsheetApp.getUi().showModalDialog(userInterface, 'Membuka WhatsApp...');
}

/**
 * 4. WEBHOOK API ENDPOINT (Menerima RSVP dari Website Frontend)
 */
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var lastRow = sheet.getLastRow();
    var dataRange = sheet.getRange(2, 1, lastRow, 8).getValues();
    
    var guestFound = false;
    
    // Cari nama tamu di kolom A dan update RSVP
    for (var i = 0; i < dataRange.length; i++) {
      if (dataRange[i][0].toLowerCase() === data.nama.toLowerCase()) {
        var targetRow = i + 2;
        sheet.getRange(targetRow, 7).setValue(data.statusRsvp); // Kolom G
        sheet.getRange(targetRow, 8).setValue(data.ucapan);    // Kolom H
        guestFound = true;
        break;
      }
    }
    
    // Jika tamu tidak terdaftar sebelumnya, tambahkan di baris baru
    if (!guestFound) {
      sheet.appendRow([data.nama, "-", "Umum / RSVP Web", "", "", "N/A", data.statusRsvp, data.ucapan]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ "result": "success" }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ "result": "error", "message": error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
